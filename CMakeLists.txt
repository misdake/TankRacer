cmake_minimum_required(VERSION 3.20)
project(TankRacerCUP VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/modules.json" MODULE_JSON)
string(JSON MODULE_COUNT LENGTH "${MODULE_JSON}")
math(EXPR MODULE_LAST "${MODULE_COUNT}-1")
foreach(INDEX RANGE ${MODULE_LAST})
  string(JSON MODULE GET "${MODULE_JSON}" ${INDEX})
  list(APPEND GAME_SOURCES "src/${MODULE}.c")
endforeach()
add_library(tankracer_core STATIC ${GAME_SOURCES})
target_include_directories(tankracer_core PUBLIC src)
add_library(tankracer_host STATIC host/game_host.c)
target_include_directories(tankracer_host PUBLIC host)
target_link_libraries(tankracer_host PUBLIC tankracer_core)
add_executable(tankracer host/native_main.c)
target_link_libraries(tankracer PRIVATE tankracer_host)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  foreach(TARGET tankracer_core tankracer_host tankracer)
    target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Werror)
  endforeach()
endif()
